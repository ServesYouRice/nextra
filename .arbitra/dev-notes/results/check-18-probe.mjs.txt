// Scratch probe for check 18 (lane depack): what does configureObsStream's
// rollback do with the stream it stopped first? Runs the same scenarios against
// the lane's current module and the original one from ac1c688.
// Usage: node depack-check18-probe.mjs <path to obsWebSocket.js> <label>
import { pathToFileURL } from 'node:url';

const [modulePath, label] = process.argv.slice(2);
const { configureObsStream } = await import(pathToFileURL(modulePath).href);

// A fake OBS with real state: the stream follows StartStream/StopStream, the
// service follows SetStreamServiceSettings, and a stream started at a WHIP
// target that refuses it stops at once.
function fakeObs({ outputActive, outputReconnecting = false, whipRefuses = false }) {
    const state = {
        outputActive,
        outputReconnecting,
        service: { streamServiceType: 'rtmp_custom', streamServiceSettings: { server: 'rtmp://live.example/app', key: 'user-key' } },
        video: { baseWidth: 1920, baseHeight: 1080, outputWidth: 1920, outputHeight: 1080, fpsNumerator: 60, fpsDenominator: 1 },
        profile: new Map(),
    };
    const log = [];
    const target = () => state.service.streamServiceSettings?.server;
    class FakeObsWebSocket {
        constructor() {
            this.closed = false;
            queueMicrotask(() => {
                this.onopen?.();
                this.onmessage?.({ data: JSON.stringify({ op: 0, d: { rpcVersion: 1 } }) });
            });
        }
        send(payload) {
            const message = JSON.parse(payload);
            if (message.op === 1) {
                queueMicrotask(() => this.onmessage?.({ data: JSON.stringify({ op: 2, d: { negotiatedRpcVersion: 1 } }) }));
                return;
            }
            if (message.op === 6) this.handle(message.d);
        }
        close() {
            if (this.closed) return;
            this.closed = true;
            log.push('close');
            this.onclose?.({ code: 1000 });
        }
        respond(request, responseData = {}, requestStatus = { result: true }) {
            queueMicrotask(() => this.onmessage?.({
                data: JSON.stringify({ op: 7, d: { requestId: request.requestId, requestType: request.requestType, requestStatus, responseData } }),
            }));
        }
        handle(request) {
            const { requestType, requestData } = request;
            switch (requestType) {
            case 'GetStreamStatus':
                log.push(requestType);
                return this.respond(request, { outputActive: state.outputActive, outputReconnecting: state.outputReconnecting });
            case 'StopStream':
                log.push(`StopStream (was ${state.outputActive ? 'active' : 'idle'} at ${target()})`);
                if (!state.outputActive) return this.respond(request, {}, { result: false, comment: 'not active' });
                state.outputActive = false;
                state.outputReconnecting = false;
                return this.respond(request);
            case 'StartStream':
                log.push(`StartStream at ${target()}`);
                if (state.outputActive) return this.respond(request, {}, { result: false, comment: 'already active' });
                state.outputActive = !(whipRefuses && state.service.streamServiceType === 'whip_custom');
                return this.respond(request);
            case 'GetStreamServiceSettings':
                log.push(requestType);
                return this.respond(request, structuredClone(state.service));
            case 'SetStreamServiceSettings':
                state.service = structuredClone(requestData);
                log.push(`SetStreamServiceSettings -> ${target()}`);
                return this.respond(request);
            case 'GetVideoSettings':
                return this.respond(request, structuredClone(state.video));
            case 'SetVideoSettings':
                log.push(requestType);
                state.video = { ...state.video, ...requestData };
                return this.respond(request);
            case 'GetProfileParameter': {
                const key = `${requestData.parameterCategory}/${requestData.parameterName}`;
                return this.respond(request, { parameterValue: state.profile.get(key) ?? null, defaultParameterValue: null });
            }
            case 'SetProfileParameter': {
                log.push(`SetProfileParameter ${requestData.parameterCategory}/${requestData.parameterName}`);
                state.profile.set(`${requestData.parameterCategory}/${requestData.parameterName}`, requestData.parameterValue);
                return this.respond(request);
            }
            case 'GetOutputList':
                return this.respond(request, { outputs: [] });
            case 'GetOutputSettings':
                return this.respond(request, {}, { result: false, comment: 'no such output' });
            case 'GetProfileList':
                return this.respond(request, { currentProfileName: 'Untitled', profiles: ['Untitled'] });
            default:
                log.push(requestType);
                return this.respond(request);
            }
        }
    }
    return { FakeObsWebSocket, log, state };
}

const setup = { whipUrl: 'http://127.0.0.1:8889/whip/ROOM', bearerToken: 'room-token' };
const scenarios = [
    {
        name: 'S0 OBS idle; setup fails (no AV1 encoder)',
        obs: { outputActive: false },
        options: { encoderSettings: { videoCodec: 'av1', obsEncoderIds: [] } },
    },
    {
        name: 'S1 OBS live at its own target; setup fails (no AV1 encoder)',
        obs: { outputActive: true },
        options: { encoderSettings: { videoCodec: 'av1', obsEncoderIds: [] } },
    },
    {
        name: 'S2 OBS live at its own target; autoStart to the Nextra WHIP target, which refuses the stream',
        obs: { outputActive: true, whipRefuses: true },
        options: { autoStart: true },
    },
    {
        name: 'S3 OBS retrying an earlier target (reconnecting); setup fails (no AV1 encoder)',
        obs: { outputActive: true, outputReconnecting: true },
        options: { encoderSettings: { videoCodec: 'av1', obsEncoderIds: [] } },
    },
    {
        name: 'S4 OBS live at its own target; the video settings are refused (bad resolution)',
        obs: { outputActive: true },
        options: { videoSettings: { outputWidth: 99999, outputHeight: 1, fpsNumerator: 60, fpsDenominator: 1 } },
        refuseVideo: true,
    },
];

console.log(`### ${label} (${modulePath})`);
for (const scenario of scenarios) {
    const obs = fakeObs(scenario.obs);
    if (scenario.refuseVideo) {
        const original = obs.FakeObsWebSocket.prototype.handle;
        obs.FakeObsWebSocket.prototype.handle = function handle(request) {
            if (request.requestType === 'SetVideoSettings') {
                obs.log.push('SetVideoSettings (refused)');
                return this.respond(request, {}, { result: false, comment: 'invalid resolution' });
            }
            return original.call(this, request);
        };
    }
    globalThis.WebSocket = obs.FakeObsWebSocket;
    const before = { active: obs.state.outputActive, target: obs.state.service.streamServiceSettings.server };
    const result = await configureObsStream({ ...setup, ...scenario.options });
    console.log(`\n${scenario.name}`);
    console.log(`  result: success=${result.success} message=${JSON.stringify(result.message)}`);
    console.log(`  requests: ${obs.log.filter((entry) => entry !== 'GetStreamStatus').join(' | ')}`);
    console.log(`  before: stream ${before.active ? 'active' : 'idle'} at ${before.target}`);
    console.log(`  after:  stream ${obs.state.outputActive ? 'active' : 'idle'} at ${obs.state.service.streamServiceSettings?.server}`);
}

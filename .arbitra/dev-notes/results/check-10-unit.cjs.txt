'use strict';

// Check 10: run the unchanged applyProducerBitrateProfile (and the helpers it calls),
// sliced out of src/HostView.jsx, against fake RTCRtpSenders.
// Usage: node socket-check-10-unit.cjs <path to HostView.jsx>

const fs = require('node:fs');
const vm = require('node:vm');

const source = fs.readFileSync(process.argv[2], 'utf8');
function slice(startMarker, endMarker) {
    const start = source.indexOf(startMarker);
    const end = source.indexOf(endMarker, start);
    if (start < 0 || end < 0) throw new Error(`marker not found: ${startMarker}`);
    return source.slice(start, end);
}
const code = [
    slice('const QUALITY_PROFILES = {', 'function loadStoredCapturePrefs'),
    slice('function getProfileRelayBitsPerSecond', 'function createGpuCapability'),
    'globalThis.api = { getSimulcastEncodings, applyProducerBitrateProfile };',
].join('\n');
const context = {};
vm.createContext(context);
vm.runInContext(code, context);
const { getSimulcastEncodings, applyProducerBitrateProfile } = context.api;

function fakeProducer(encodings) {
    let params = { encodings: encodings.map((e) => ({ ...e })) };
    return {
        rtpSender: {
            getParameters: () => JSON.parse(JSON.stringify(params)),
            setParameters: async (next) => { params = JSON.parse(JSON.stringify(next)); },
            current: () => params,
        },
    };
}

(async () => {
    const produced = getSimulcastEncodings('1440p', 60);
    console.log('produce() encodings for 1440p/60:', JSON.stringify(produced));

    // What the app's produce() asks for, and what Chromium/Firefox keep (see the browser probe).
    const three = fakeProducer(produced.map(({ rid, scaleResolutionDownBy }) => ({ rid, scaleResolutionDownBy, maxBitrate: 1 })));
    await applyProducerBitrateProfile(three, '1440p', 60);
    console.log('3 encodings r0,r1,r2 ->', JSON.stringify(three.rtpSender.current().encodings));

    // The claim's premise: a sender with one encoding. A browser can only end up there by
    // trimming the requested sendEncodings from the tail (webrtc-pc addTransceiver, maxN = 1)
    // or keeping only the first one when the answer rejects simulcast, so the lone encoding is
    // the requested r0 (scaleResolutionDownBy 4) with its rid removed.
    const lone = fakeProducer([{ scaleResolutionDownBy: 4, maxBitrate: produced[0].maxBitrate }]);
    await applyProducerBitrateProfile(lone, '1440p', 60);
    console.log('1 encoding (trimmed r0) ->', JSON.stringify(lone.rtpSender.current().encodings));
})();

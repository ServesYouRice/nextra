'use strict';
// Check 08 (C-52a422ba4d2d) probe: can a WHIP WebRtcTransport that the server closed itself
// (session replacement, DELETE, aborted startup) still run the 'failed'/'closed' branch of the
// POST handler's dtlsstatechange listener?
//
// Real mediasoup worker, router and WebRtcTransports (UDP on 127.0.0.1), the real WHIP router on a
// real HTTP server. DTLS never really starts (no ICE client), so a DTLS state change is put in the
// way mediasoup's worker delivers one: through Channel.processNotification(), which dispatches it
// with setImmediate to the listener registered for the transport id. That reaches mediasoup's own
// WebRtcTransport notification handler and its safeEmit('dtlsstatechange', ...).
//
// Usage: node lifecycle-check-08-probe.cjs <repo root>

const path = require('node:path');
const { createRequire } = require('node:module');

const root = path.resolve(process.argv[2] || 'C:/Users/V/nextra-lanes/lifecycle');
const rootRequire = createRequire(path.join(root, 'server.js'));

const express = rootRequire('express');
const mediasoup = rootRequire('mediasoup');
const http = require('node:http');
const config = rootRequire('./config');
const whipRoutes = rootRequire('./lib/whipRoutes');
const { createRoom, destroyRoom } = rootRequire('./lib/rooms');
const { Event } = rootRequire('./node_modules/mediasoup/node/lib/fbs/notification/event.js');
const { DtlsState } = rootRequire('./node_modules/mediasoup/node/lib/fbs/web-rtc-transport/dtls-state.js');
const msVersion = JSON.parse(require('node:fs').readFileSync(path.join(root, 'node_modules/mediasoup/package.json'), 'utf8')).version;

const FINGERPRINT = Array.from({ length: 32 }, (_, i) => (i * 7 + 17).toString(16).padStart(2, '0').toUpperCase()).join(':');
const OFFER = [
    'v=0',
    'o=- 0 0 IN IP4 127.0.0.1',
    's=-',
    't=0 0',
    'a=group:BUNDLE 0',
    'a=ice-ufrag:testufrag',
    'a=ice-pwd:testpwd1234567890123456',
    `a=fingerprint:sha-256 ${FINGERPRINT}`,
    'a=setup:actpass',
    'm=video 9 UDP/TLS/RTP/SAVPF 96',
    'c=IN IP4 0.0.0.0',
    'a=mid:0',
    'a=sendonly',
    'a=rtpmap:96 H264/90000',
    'a=fmtp:96 profile-level-id=42e01f;packetization-mode=1;level-asymmetry-allowed=1',
    'a=ssrc:1111 cname:test',
    '',
].join('\r\n');

const GRACE_MS = 150;
const settle = (ms) => new Promise((resolve) => setTimeout(resolve, ms));

const hostMetrics = [];
whipRoutes.setMediaLifecycle({
    startFallbackRelay: async () => {},
    stopFallbackRelay: (room) => { if (room) room.fallbackWorker = null; },
    emitHostMetrics: (_io, code) => hostMetrics.push(code),
});
whipRoutes.setIo({ to: () => ({ emit() {} }) });

const transportLog = new Map();
let connectHook = null;

function makeFactory() {
    return async (router) => {
        const transport = await router.createWebRtcTransport({
            listenInfos: [{ protocol: 'udp', ip: '127.0.0.1', portRange: { min: 47000, max: 47999 } }],
        });
        const log = [];
        transportLog.set(transport, log);
        transport.on('dtlsstatechange', (state) => log.push(`transport:dtlsstatechange:${state}`));
        transport.observer.on('dtlsstatechange', (state) => log.push(`observer:dtlsstatechange:${state}`));
        transport.on('routerclose', () => log.push('transport:routerclose'));
        transport.observer.on('close', () => log.push('observer:close'));
        if (connectHook) {
            const hook = connectHook;
            connectHook = null;
            const connect = transport.connect.bind(transport);
            transport.connect = async (params) => { await hook(transport); return connect(params); };
        }
        return {
            transport,
            params: {
                id: transport.id,
                iceParameters: transport.iceParameters,
                iceCandidates: transport.iceCandidates,
                dtlsParameters: transport.dtlsParameters,
            },
        };
    };
}

// Deliver a DTLS state change exactly as mediasoup's worker does (Channel.processNotification).
function deliverDtls(transport, state) {
    const fbsState = { connected: DtlsState.CONNECTED, failed: DtlsState.FAILED, closed: DtlsState.CLOSED }[state];
    transport.channelForTesting.processNotification({
        handlerId: () => transport.id,
        event: () => Event.WEBRTCTRANSPORT_DTLS_STATE_CHANGE,
        body: (notification) => {
            notification.dtlsState = () => fbsState;
            notification.remoteCert = () => null;
            return notification;
        },
    });
}

function snapshot(room) {
    return {
        whipSessionId: room.whipSessionId,
        whipResourceId: room.whipResourceId,
        whipTransportId: room.whipTransport?.id ?? null,
        whipTransportClosed: room.whipTransport ? room.whipTransport.closed : null,
        whipConnected: room.whipConnected,
        whipReconnecting: room.whipReconnecting,
        whipGraceTimerSet: Boolean(room.whipGraceTimer),
        whipStarting: room.whipStarting,
    };
}

async function main() {
    config.WHIP_ENABLED = true;
    config.WHIP_GRACE_TIMEOUT_MS = GRACE_MS;

    const worker = await mediasoup.createWorker({ logLevel: 'error' });
    const router = await worker.createRouter({
        mediaCodecs: [
            { kind: 'video', mimeType: 'video/H264', clockRate: 90000, parameters: { 'packetization-mode': 1, 'profile-level-id': '42e01f', 'level-asymmetry-allowed': 1 } },
            { kind: 'audio', mimeType: 'audio/opus', clockRate: 48000, channels: 2 },
        ],
    });

    const app = express();
    app.use('/whip', whipRoutes.createWhipRouter(router, { createWebRtcTransport: makeFactory() }));
    const server = http.createServer(app);
    await new Promise((resolve) => server.listen(0, '127.0.0.1', resolve));
    const base = `http://127.0.0.1:${server.address().port}`;

    const post = (room, signal) => fetch(`${base}/whip/broadcast/${room.code}`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/sdp', Authorization: `Bearer ${room.hostToken}` },
        body: OFFER,
        signal,
    });
    const del = (location) => fetch(location, { method: 'DELETE' });
    const listeners = (transport) => transport.channelForTesting.listenerCount(transport.id);

    const report = { root, mediasoup: msVersion, graceMs: GRACE_MS, scenarios: {} };
    const rooms = [];
    const newRoom = async (name) => { const room = await createRoom(`check08-${name}`, { ingestMode: 'obs' }); rooms.push(room); return room; };

    try {
        // A. mediasoup contract: a closed WebRtcTransport.
        {
            const { transport } = await makeFactory()(router);
            await transport.connect({ dtlsParameters: { role: 'client', fingerprints: [{ algorithm: 'sha-256', value: FINGERPRINT }] } });
            const before = listeners(transport);
            transport.close();
            const after = listeners(transport);
            deliverDtls(transport, 'closed');
            await settle(30);
            report.scenarios.A_mediasoupClose = {
                channelListenersBeforeClose: before,
                channelListenersAfterClose: after,
                dtlsStateAfterClose: transport.dtlsState,
                events: transportLog.get(transport),
            };
        }

        // B. Control: the same delivery on the room's current transport does reach the handler.
        {
            const room = await newRoom('control');
            const res = await post(room);
            const t = room.whipTransport;
            const metricsBefore = hostMetrics.length;
            deliverDtls(t, 'closed');
            await settle(20);
            const afterEvent = snapshot(room);
            await settle(GRACE_MS + 100);
            report.scenarios.B_controlCurrentTransport = {
                status: res.status,
                events: transportLog.get(t),
                afterEvent,
                hostMetricsCalls: hostMetrics.length - metricsBefore,
                afterGrace: snapshot(room),
                transportClosedAfterGrace: t.closed,
            };
        }

        // C. Replacement POST, then DTLS 'closed' and 'failed' delivered to the replaced transport.
        {
            const room = await newRoom('replace');
            const first = await post(room);
            const t1 = room.whipTransport;
            const session1 = room.whipSessionId;
            const t1ListenersBefore = listeners(t1);
            const metricsBefore = hostMetrics.length;
            const second = await post(room);
            const t2 = room.whipTransport;
            const session2 = room.whipSessionId;
            deliverDtls(t1, 'closed');
            deliverDtls(t1, 'failed');
            await settle(20);
            const afterEvents = snapshot(room);
            await settle(GRACE_MS + 100);
            report.scenarios.C_replacement = {
                statuses: [first.status, second.status],
                t1Closed: t1.closed,
                t1ChannelListeners: [t1ListenersBefore, listeners(t1)],
                t1Events: transportLog.get(t1),
                newSessionIsSecond: session2 !== session1 && room.whipSessionId === session2,
                afterEvents,
                afterGrace: snapshot(room),
                t2StillCurrentAndOpen: room.whipTransport === t2 && !t2.closed,
                hostMetricsCalls: hostMetrics.length - metricsBefore,
            };
        }

        // D. DELETE, then DTLS 'closed' and 'failed' delivered to the deleted session's transport.
        {
            const room = await newRoom('delete');
            const res = await post(room);
            const t = room.whipTransport;
            const metricsBefore = hostMetrics.length;
            const deleted = await del(res.headers.get('location'));
            deliverDtls(t, 'closed');
            deliverDtls(t, 'failed');
            await settle(20);
            const afterEvents = snapshot(room);
            await settle(GRACE_MS + 100);
            report.scenarios.D_delete = {
                statuses: [res.status, deleted.status],
                transportClosed: t.closed,
                channelListeners: listeners(t),
                events: transportLog.get(t),
                afterEvents,
                afterGrace: snapshot(room),
                hostMetricsCalls: hostMetrics.length - metricsBefore,
            };
        }

        // E. A notification already queued by the channel when the session is closed
        //    (closeWhipSession is what DELETE and the replacing POST run, synchronously).
        {
            const room = await newRoom('inflight');
            await post(room);
            const t = room.whipTransport;
            const metricsBefore = hostMetrics.length;
            deliverDtls(t, 'closed');
            whipRoutes.closeWhipSession(room);
            await settle(20);
            const afterEvents = snapshot(room);
            await settle(GRACE_MS + 100);
            report.scenarios.E_inFlightAtClose = {
                transportClosed: t.closed,
                events: transportLog.get(t),
                afterEvents,
                afterGrace: snapshot(room),
                hostMetricsCalls: hostMetrics.length - metricsBefore,
            };
        }

        // F. Aborted startup: the client goes away while transport.connect() is pending, after the
        //    dtlsstatechange listener is registered; then DTLS 'closed' is delivered.
        {
            const room = await newRoom('abort');
            const controller = new AbortController();
            let aborted;
            connectHook = async (transport) => {
                aborted = transport;
                controller.abort();
                for (let i = 0; i < 200 && !transport.closed; i += 1) await settle(5);
            };
            const metricsBefore = hostMetrics.length;
            const outcome = await post(room, controller.signal).then((r) => `status ${r.status}`, (err) => err.name);
            await settle(50);
            const handlerListeners = aborted.listenerCount('dtlsstatechange');
            deliverDtls(aborted, 'closed');
            await settle(20);
            const afterEvents = snapshot(room);
            await settle(GRACE_MS + 100);
            report.scenarios.F_abortedStartup = {
                clientOutcome: outcome,
                transportClosed: aborted.closed,
                whipHandlerAttached: handlerListeners > 1,
                channelListeners: listeners(aborted),
                events: transportLog.get(aborted),
                afterEvents,
                afterGrace: snapshot(room),
                hostMetricsCalls: hostMetrics.length - metricsBefore,
            };
        }

        // G. Premise check (not mediasoup's behaviour): if close() did emit dtlsstatechange 'closed'
        //    first, as the claim assumes, the replacement session would get a grace timer from the
        //    old transport's unguarded branch and be closed when it expires.
        {
            const room = await newRoom('premise');
            await post(room);
            const t1 = room.whipTransport;
            const realClose = t1.close.bind(t1);
            t1.close = () => { t1.emit('dtlsstatechange', 'closed'); realClose(); };
            const metricsBefore = hostMetrics.length;
            await post(room);
            const session2 = room.whipSessionId;
            const t2 = room.whipTransport;
            const afterReplace = snapshot(room);
            await settle(GRACE_MS + 100);
            report.scenarios.G_premiseIfCloseEmittedClosed = {
                afterReplace,
                hostMetricsCalls: hostMetrics.length - metricsBefore,
                afterGrace: snapshot(room),
                replacementKilled: room.whipSessionId === null && session2 !== null && t2.closed,
            };
        }
    } finally {
        for (const room of rooms) { try { whipRoutes.closeWhipSession(room); } catch { } destroyRoom(room.code); }
        await new Promise((resolve) => server.close(resolve));
        router.close();
        worker.close();
    }

    const s = report.scenarios;
    const quiet = (sc) => sc.hostMetricsCalls === 0 && !sc.afterEvents.whipReconnecting && !sc.afterEvents.whipGraceTimerSet
        && !sc.afterGrace.whipReconnecting && !sc.afterGrace.whipGraceTimerSet
        && !sc.afterEvents.whipConnected
        && (sc.events || sc.t1Events).every((e) => !e.includes('dtlsstatechange'));
    report.verdict = {
        mediasoupDropsNotificationsAfterClose: s.A_mediasoupClose.channelListenersAfterClose === 0
            && s.A_mediasoupClose.events.every((e) => !e.includes('dtlsstatechange')),
        controlReachesHandler: s.B_controlCurrentTransport.afterEvent.whipReconnecting === true
            && s.B_controlCurrentTransport.afterEvent.whipGraceTimerSet === true
            && s.B_controlCurrentTransport.hostMetricsCalls === 1
            && s.B_controlCurrentTransport.afterGrace.whipSessionId === null
            && s.B_controlCurrentTransport.transportClosedAfterGrace === true,
        replacementUntouched: quiet(s.C_replacement) && s.C_replacement.t2StillCurrentAndOpen && s.C_replacement.newSessionIsSecond
            && s.C_replacement.afterGrace.whipSessionId !== null,
        deleteUntouched: quiet(s.D_delete) && s.D_delete.afterGrace.whipSessionId === null,
        inFlightUntouched: quiet(s.E_inFlightAtClose),
        abortedStartupUntouched: quiet(s.F_abortedStartup) && s.F_abortedStartup.whipHandlerAttached,
    };
    report.claimHolds = !Object.values(report.verdict).every(Boolean);
    const out = JSON.stringify(report, null, 2);
    if (process.argv[3]) require('node:fs').writeFileSync(process.argv[3], `${out}\n`);
    console.log(out);
    process.exitCode = report.claimHolds ? 1 : 0;
}

main().catch((err) => { console.error(err); process.exitCode = 2; });

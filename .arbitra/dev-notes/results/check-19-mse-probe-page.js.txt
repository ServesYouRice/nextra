// Runs inside the browser page. Records a relay stream the way HostView does
// (canvas + silent audio, MediaRecorder with a timeslice, a recorder restart =
// a new generation), then feeds it to a copy of WatchView's relay player logic.
window.recordRelay = async function recordRelay({ gen1Mime, gen2Mime, timeslice = 300, gen1Ms = 3000, gen2Ms = 4000, bitrate = 4_000_000 }) {
    const canvas = document.createElement('canvas');
    canvas.width = 1280;
    canvas.height = 720;
    document.body.appendChild(canvas);
    const ctx = canvas.getContext('2d');
    let frame = 0;
    let drawing = true;
    const draw = () => {
        if (!drawing) return;
        frame += 1;
        ctx.fillStyle = `hsl(${(frame * 7) % 360} 80% 45%)`;
        ctx.fillRect(0, 0, canvas.width, canvas.height);
        for (let i = 0; i < 60; i += 1) {
            ctx.fillStyle = `hsl(${(frame * 13 + i * 37) % 360} 70% ${30 + (i % 5) * 10}%)`;
            ctx.fillRect((i * 97 + frame * 11) % 1280, (i * 53 + frame * 5) % 720, 80, 60);
        }
        ctx.fillStyle = 'white';
        ctx.font = '64px sans-serif';
        ctx.fillText(`frame ${frame}`, 40, 100);
        setTimeout(draw, 33);
    };
    draw();
    const stream = canvas.captureStream(30);
    // HostView adds a silent audio track when the capture has none.
    const audioCtx = new AudioContext();
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    gain.gain.value = 0;
    const dst = audioCtx.createMediaStreamDestination();
    osc.connect(gain);
    gain.connect(dst);
    osc.start();
    stream.addTrack(dst.stream.getAudioTracks()[0]);

    const t0 = performance.now();
    const events = [];
    const pending = [];
    const startRecorder = (mime, generation) => {
        const recorder = new MediaRecorder(stream, { mimeType: mime, videoBitsPerSecond: bitrate });
        recorder.onstart = () => {
            events.push({ at: performance.now() - t0, type: 'init', generation, mimeType: recorder.mimeType || mime });
        };
        recorder.ondataavailable = (evt) => {
            if (!evt.data || evt.data.size <= 0) return;
            const slot = { at: performance.now() - t0, type: 'chunk', generation, data: null };
            events.push(slot);
            pending.push(evt.data.arrayBuffer().then((buf) => { slot.data = buf; }));
        };
        const stopped = new Promise((resolve) => { recorder.onstop = resolve; });
        recorder.start(timeslice);
        return { recorder, stopped };
    };
    const g1 = startRecorder(gen1Mime, 1);
    await new Promise((r) => setTimeout(r, gen1Ms));
    // HostView: stopRelayRecorder(); startRelayRecorder();
    g1.recorder.stop();
    const g2 = startRecorder(gen2Mime, 2);
    await new Promise((r) => setTimeout(r, gen2Ms));
    g2.recorder.stop();
    await Promise.all([g1.stopped, g2.stopped]);
    await Promise.all(pending);
    drawing = false;
    osc.stop();
    await audioCtx.close();
    stream.getTracks().forEach((t) => t.stop());
    canvas.remove();
    window.__relayEvents = events;
    return events.map((e) => ({ at: Math.round(e.at), type: e.type, generation: e.generation, mimeType: e.mimeType, size: e.data ? e.data.byteLength : undefined }));
};

window.getRelayBytes = function getRelayBytes(generation) {
    // base64 of the concatenated chunks of a generation, for offline parsing
    const parts = window.__relayEvents.filter((e) => e.type === 'chunk' && e.generation === generation).map((e) => new Uint8Array(e.data));
    const total = parts.reduce((n, p) => n + p.length, 0);
    const all = new Uint8Array(total);
    let off = 0;
    const sizes = [];
    for (const p of parts) { all.set(p, off); off += p.length; sizes.push(p.length); }
    let bin = '';
    for (let i = 0; i < all.length; i += 0x8000) bin += String.fromCharCode.apply(null, all.subarray(i, i + 0x8000));
    return { b64: btoa(bin), sizes };
};

// The viewer: a copy of WatchView.jsx startRelayPlayback's queue, onChunk,
// onMediaInit and SourceBuffer handling (lines 412-668), fed with the recorded
// events through a server that drops chunks of a superseded generation
// (lib/socket.js media-chunk / media-init handlers).
// mode 'current' = the code as it is; 'fixed' = abort()/changeType() on a new
// generation. burstMs: events within [switchAt - burstMs, switchAt + 50] are
// delivered in one task, as a socket backlog would be.
window.playRelay = async function playRelay({ mode = 'current', burstMs = 0, settleMs = 2500 }) {
    const events = window.__relayEvents;
    const video = document.createElement('video');
    video.muted = true;
    video.autoplay = true;
    document.body.appendChild(video);

    const log = [];
    let chunkQueue = [];
    let queuedBytes = 0;
    let appendingLock = false;
    let sourceBuffer = null;
    let mediaSource = null;
    let selectedGeneration = null;
    let mimeType = null;
    let sbErrors = 0;
    let appended = { 1: 0, 2: 0 };
    let flushed = 0;
    let serverGeneration = null;

    const processQueue = () => {
        if (appendingLock || !sourceBuffer || sourceBuffer.updating) return;
        if (!chunkQueue.length) return;
        if (mediaSource?.readyState !== 'open') return;
        try {
            appendingLock = true;
            const next = chunkQueue.shift();
            queuedBytes -= next.u8.byteLength;
            appended[next.generation] += 1;
            sourceBuffer.appendBuffer(next.u8);
        } catch (err) {
            log.push(`appendBuffer threw ${err.name}: ${err.message}`);
            appendingLock = false;
        }
    };
    const enqueueChunk = (u8, generation) => {
        chunkQueue.push({ u8, generation });
        queuedBytes += u8.byteLength;
        if (sourceBuffer && mediaSource?.readyState === 'open' && !appendingLock && !sourceBuffer.updating) processQueue();
    };
    const onChunk = (payload) => {
        if (payload.generation !== selectedGeneration) return;
        enqueueChunk(new Uint8Array(payload.chunk), payload.generation);
    };
    const onMediaInit = (payload) => {
        if (chunkQueue.length > 0) {
            flushed += chunkQueue.length;
            chunkQueue.length = 0;
            queuedBytes = 0;
        }
        if (mode === 'fixed' && sourceBuffer && mediaSource?.readyState === 'open') {
            try {
                // Reset the parser: bytes of the old generation's last,
                // possibly partial, element must not run into the new init.
                sourceBuffer.abort();
                appendingLock = false;
                if (payload.mimeType !== mimeType && typeof sourceBuffer.changeType === 'function') {
                    sourceBuffer.changeType(payload.mimeType);
                }
            } catch (err) {
                log.push(`fix threw ${err.name}: ${err.message}`);
            }
        }
        selectedGeneration = payload.generation;
        mimeType = payload.mimeType;
    };

    // Server
    const serverDeliver = (ev) => {
        if (ev.type === 'init') {
            if (serverGeneration != null && ev.generation <= serverGeneration) return;
            serverGeneration = ev.generation;
            log.push(`init gen ${ev.generation} ${ev.mimeType} at ${Math.round(ev.at)}`);
            onMediaInit({ generation: ev.generation, mimeType: ev.mimeType });
            return;
        }
        if (ev.generation !== serverGeneration) { log.push(`server dropped late gen ${ev.generation} chunk`); return; }
        onChunk({ generation: ev.generation, chunk: ev.data });
    };

    // Bootstrap as WatchView does for the first generation.
    const firstInit = events.find((e) => e.type === 'init' && e.generation === 1);
    serverGeneration = 1;
    selectedGeneration = 1;
    mimeType = firstInit.mimeType;
    mediaSource = new MediaSource();
    video.src = URL.createObjectURL(mediaSource);
    await new Promise((r) => mediaSource.addEventListener('sourceopen', r, { once: true }));
    mediaSource.addEventListener('sourceended', () => log.push(`sourceended (readyState ${mediaSource.readyState})`));
    sourceBuffer = mediaSource.addSourceBuffer(mimeType);
    if (sourceBuffer.mode === 'segments') sourceBuffer.mode = 'sequence';
    sourceBuffer.addEventListener('error', () => { sbErrors += 1; log.push('SourceBuffer error event'); appendingLock = false; });
    sourceBuffer.addEventListener('updateend', () => { appendingLock = false; processQueue(); });
    video.addEventListener('error', () => log.push(`video error ${video.error?.code} ${video.error?.message}`));
    video.play().catch(() => {});

    const switchAt = events.find((e) => e.type === 'init' && e.generation === 2).at;
    const rest = events.filter((e) => !(e.type === 'init' && e.generation === 1));
    const t0 = performance.now();
    let i = 0;
    while (i < rest.length) {
        const ev = rest[i];
        const inBurst = burstMs > 0 && ev.at >= switchAt - burstMs && ev.at <= switchAt + 50;
        if (inBurst) {
            // Wait until the end of the burst window, then deliver all of it in one task.
            const end = switchAt + 50;
            const wait = end - (performance.now() - t0);
            if (wait > 0) await new Promise((r) => setTimeout(r, wait));
            while (i < rest.length && rest[i].at >= switchAt - burstMs && rest[i].at <= end) { serverDeliver(rest[i]); i += 1; }
            continue;
        }
        const wait = ev.at - (performance.now() - t0);
        if (wait > 0) await new Promise((r) => setTimeout(r, wait));
        serverDeliver(ev);
        i += 1;
    }
    await new Promise((r) => setTimeout(r, settleMs));
    const ranges = [];
    try { for (let k = 0; k < sourceBuffer.buffered.length; k += 1) ranges.push([+sourceBuffer.buffered.start(k).toFixed(2), +sourceBuffer.buffered.end(k).toFixed(2)]); } catch (err) { ranges.push(`buffered threw ${err.name}`); }
    const result = {
        mode, burstMs,
        sbErrors,
        mediaSourceState: mediaSource.readyState,
        videoError: video.error ? `${video.error.code} ${video.error.message}` : null,
        buffered: ranges,
        currentTime: +video.currentTime.toFixed(2),
        appended, flushed, queueLeft: chunkQueue.length,
        log,
    };
    video.remove();
    return result;
};

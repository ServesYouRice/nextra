// Global setup for check 12. Runs after Playwright has cleared outputDir.
// CHECK12_BLOCKER=1 plants a ticket that sorts before any ticket the spec can
// write (its names start with Date.now()), i.e. a holder that is still inside
// its test. CHECK12_STALE_DIR plants the same kind of ticket in another folder
// before the run, standing for one left behind by an earlier, killed run.
import fs from 'node:fs';
import path from 'node:path';

export const BLOCKER = '0000000000001-9999-00000001-0001.ticket';

function log(line) {
    const file = process.env.CHECK12_LOG;
    if (file) fs.appendFileSync(file, `[setup] ${line}\n`);
}

export default async function globalSetup() {
    const lockDir = process.env.CHECK12_LOCK_DIR;
    if (process.env.CHECK12_BLOCKER === '1') {
        fs.mkdirSync(lockDir, { recursive: true });
        fs.writeFileSync(path.join(lockDir, BLOCKER), '');
        log(`planted holder ticket ${BLOCKER} in ${lockDir}`);
    }
    for (const dir of [lockDir, process.env.CHECK12_STALE_DIR].filter(Boolean)) {
        let entries;
        try { entries = fs.readdirSync(dir).sort(); } catch { entries = '(no folder)'; }
        log(`at run start ${dir}: ${JSON.stringify(entries)}`);
    }
}

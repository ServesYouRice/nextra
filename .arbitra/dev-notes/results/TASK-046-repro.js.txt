'use strict';
// Ad hoc reproduction of the TASK-046 defect against the UNCHANGED package-app.js.
// Loads the packager with process.platform faked as darwin, a manifest whose
// darwin-arm64 digest is the digest of a small fake .tgz, and the stage directory
// moved into the scratchpad. Then runs bundleCloudflared() with CLOUDFLARED_PATH
// pointing at the fake archive and reports what landed in the stage.
const fs = require('node:fs');
const os = require('node:os');
const path = require('node:path');
const crypto = require('node:crypto');
const Module = require('node:module');
const { spawnSync } = require('node:child_process');

const repo = process.argv[2];
const work = fs.mkdtempSync(path.join(os.tmpdir(), 'package-TASK-046-repro-'));
const scripts = path.join(repo, 'scripts');

// A fake archive holding one `cloudflared` entry (a shell script standing in for the Mach-O).
const payloadDir = path.join(work, 'payload');
fs.mkdirSync(payloadDir);
fs.writeFileSync(path.join(payloadDir, 'cloudflared'), '#!/bin/sh\necho fake cloudflared\n');
const archive = path.join(work, 'cloudflared-darwin-arm64.tgz');
const tarResult = spawnSync('tar', ['-czf', '../cloudflared-darwin-arm64.tgz', 'cloudflared'], { encoding: 'utf8', cwd: payloadDir });
if (tarResult.status !== 0) throw new Error(`tar failed: ${tarResult.stderr}`);
const archiveSha = crypto.createHash('sha256').update(fs.readFileSync(archive)).digest('hex');

Object.defineProperty(process, 'platform', { value: 'darwin' });
Object.defineProperty(process, 'arch', { value: 'arm64' });

const manifestPath = path.join(scripts, 'cloudflared-manifest.json');
const manifest = JSON.parse(fs.readFileSync(manifestPath, 'utf8'));
manifest.assets['cloudflared-darwin-arm64.tgz'] = archiveSha;
Module._cache[manifestPath] = { id: manifestPath, filename: manifestPath, loaded: true, exports: manifest, children: [], paths: [] };

const stage = path.join(work, 'stage');
fs.mkdirSync(stage, { recursive: true });
const file = path.join(scripts, 'package-app.js');
const source = fs.readFileSync(file, 'utf8')
    .replace("const stageDir = path.join(projectRoot, '.caxa-stage');", `const stageDir = ${JSON.stringify(stage)};`)
    + '\nmodule.exports.__bundleCloudflared = bundleCloudflared;\n';
const mod = new Module(file, null);
mod.filename = file;
mod.paths = Module._nodeModulePaths(scripts);
mod._compile(source, file);

process.env.CLOUDFLARED_PATH = archive;
delete process.env.ALLOW_LOCAL_CLOUDFLARED;

(async () => {
    try {
        await mod.exports.__bundleCloudflared();
    } catch (err) {
        console.log('bundleCloudflared threw:', err.message);
    }
    const staged = path.join(stage, 'cloudflared');
    if (fs.existsSync(staged)) {
        const bytes = fs.readFileSync(staged);
        const sameAsArchive = bytes.equals(fs.readFileSync(archive));
        console.log(`staged cloudflared exists: ${bytes.length} bytes, identical to the .tgz archive: ${sameAsArchive}`);
        console.log(`first bytes: ${bytes.subarray(0, 2).toString('hex')} (1f8b = gzip magic)`);
        console.log(`mode: ${(fs.statSync(staged).mode & 0o777).toString(8)}`);
    } else {
        console.log('nothing staged');
    }
    fs.rmSync(work, { recursive: true, force: true });
})();

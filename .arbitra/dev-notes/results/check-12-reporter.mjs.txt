// Reporter for check 12: logs each test's outcome and the lock folder(s) right
// after the test (its afterEach and worker cleanup have run by then), and when
// CHECK12_BLOCKER=1 removes the planted holder ticket as the first test ends.
import fs from 'node:fs';
import path from 'node:path';

const BLOCKER = '0000000000001-9999-00000001-0001.ticket';
const ANSI = /\u001b\[[0-9;]*m/g;

export default class Check12Reporter {
    constructor() {
        this.started = Date.now();
        this.ended = 0;
        this.dirs = [process.env.CHECK12_LOCK_DIR, process.env.CHECK12_STALE_DIR].filter(Boolean);
    }

    printsToStdio() {
        return false;
    }

    write(line) {
        const seconds = ((Date.now() - this.started) / 1000).toFixed(1).padStart(6);
        const text = `[+${seconds}s] ${line}\n`;
        if (process.env.CHECK12_LOG) fs.appendFileSync(process.env.CHECK12_LOG, text);
    }

    listing() {
        return this.dirs.map((dir) => {
            let entries;
            try { entries = fs.readdirSync(dir).sort(); } catch { entries = '(no folder)'; }
            return `${dir} = ${JSON.stringify(entries)}`;
        }).join(' ; ');
    }

    onBegin(_config, suite) {
        this.write(`run begins: ${suite.allTests().length} tests; ${this.listing()}`);
    }

    onTestBegin(test) {
        this.write(`begin  "${test.title}"`);
    }

    onTestEnd(test, result) {
        this.ended += 1;
        const error = (result.error?.message || '').replace(ANSI, '').split('\n')[0];
        this.write(`end #${String(this.ended).padStart(2)} ${result.status} after ${result.duration} ms "${test.title}"`
            + `${error ? ` | ${error}` : ''}`);
        this.write(`       lock folder now: ${this.listing()}`);
        if (this.ended === 1 && process.env.CHECK12_BLOCKER === '1') {
            fs.rmSync(path.join(process.env.CHECK12_LOCK_DIR, BLOCKER), { force: true });
            this.write(`       holder ticket ${BLOCKER} removed (the holder released the lock)`);
        }
    }

    onEnd(result) {
        this.write(`run ends: ${result.status}; ${this.listing()}`);
    }
}

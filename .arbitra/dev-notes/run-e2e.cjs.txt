'use strict';

// Runs Playwright for ONE nextra worktree on its own ports, without the shared
// lock: builds the client into a private temp folder, claims a free port pair
// the way the real-server tests do (tests/serverPorts.js), starts server.js on
// it with the same environment playwright.config.mjs gives its webServer, runs
// `playwright test <your arguments>` with NEXTRA_E2E_PORT and
// NEXTRA_E2E_RTC_MIN_PORT / NEXTRA_E2E_RTC_MAX_PORT set to that pair, then stops
// the server and removes the build. Playwright's own web server never starts
// (it reuses the running one), so nothing is built into the linked dist.
//
// Usage, from the worktree root:
//   node C:/Users/V/nextra/.arbitra/dev-notes/run-e2e.cjs [playwright test arguments, e.g. tests/browser/ui-semantics.spec.mjs]

const { spawn, spawnSync } = require('node:child_process');
const fs = require('node:fs');
const https = require('node:https');
const os = require('node:os');
const path = require('node:path');

const root = process.cwd();
const lane = path.basename(root);
if (!fs.existsSync(path.join(root, 'playwright.config.mjs'))) {
    console.error(`run-e2e: ${root} has no playwright.config.mjs; run this from a worktree root.`);
    process.exit(2);
}

const { claimServerPorts } = require(path.join(root, 'tests', 'serverPorts.js'));

// The build, the certificate and the server's files go in a fresh folder under
// NEXTRA_E2E_WORK_DIR (your scratchpad) or the system temp folder.
const workDir = fs.mkdtempSync(path.join(process.env.NEXTRA_E2E_WORK_DIR || os.tmpdir(), `nextra-e2e-${lane}-`));
const distDir = path.join(workDir, 'dist');
const certDir = path.join(workDir, 'certs');

function sleep(ms) {
    return new Promise((resolve) => setTimeout(resolve, ms));
}

function isReady(port) {
    return new Promise((resolve) => {
        const req = https.request({
            host: '127.0.0.1', port, path: '/readyz', method: 'GET', rejectUnauthorized: false, timeout: 2000,
        }, (res) => {
            res.resume();
            resolve(res.statusCode === 200);
        });
        req.on('error', () => resolve(false));
        req.on('timeout', () => { req.destroy(); resolve(false); });
        req.end();
    });
}

function killTree(child) {
    if (!child || child.exitCode !== null) return;
    if (process.platform === 'win32') {
        spawnSync('taskkill', ['/PID', String(child.pid), '/T', '/F'], { windowsHide: true });
    } else {
        child.kill('SIGKILL');
    }
}

async function main() {
    const build = spawnSync(process.execPath, [
        path.join(root, 'node_modules', 'vite', 'bin', 'vite.js'),
        'build', '--outDir', distDir, '--emptyOutDir',
    ], { cwd: root, stdio: 'inherit', windowsHide: true });
    if (build.status !== 0) throw new Error(`vite build failed (status ${build.status})`);

    const { port, rtcPort } = await claimServerPorts();
    console.log(`run-e2e: lane ${lane}, HTTPS port ${port}, RTC port ${rtcPort}, build ${distDir}`);

    let output = '';
    const server = spawn(process.execPath, ['server.js'], {
        cwd: root,
        windowsHide: true,
        stdio: ['ignore', 'pipe', 'pipe'],
        env: {
            ...process.env,
            NODE_ENV: 'test',
            PORT: String(port),
            BIND_HOST: '127.0.0.1',
            LOCAL_HTTPS: 'true',
            HTTPS_CERT_DIR: certDir,
            OPEN_BROWSER: 'false',
            AUTO_PUBLIC_TUNNEL: 'false',
            SHARE_BASE_URL: `https://nextra.cloudflare.test:${port}`,
            WHIP_ENABLED: 'false',
            WHEP_ENABLED: 'false',
            RTC_LISTEN_IP: '127.0.0.1',
            RTC_MIN_PORT: String(rtcPort),
            RTC_MAX_PORT: String(rtcPort),
            NEXTRA_SMOKE_TEST: '1',
            NEXTRA_TEST_DIST_DIR: distDir,
        },
    });
    server.stdout.on('data', (chunk) => { output += chunk.toString(); });
    server.stderr.on('data', (chunk) => { output += chunk.toString(); });

    try {
        const deadline = Date.now() + 60_000;
        for (;;) {
            if (server.exitCode !== null) throw new Error(`server exited early (code ${server.exitCode})\n${output}`);
            if (Date.now() > deadline) throw new Error(`server was not ready in time\n${output}`);
            if (await isReady(port)) break;
            await sleep(200);
        }

        // Traces and videos go to this run's own folder (kept when the run fails),
        // so two runs from one worktree do not wipe each other's test-results.
        const args = process.argv.slice(2);
        if (!args.some((arg) => arg === '--output' || arg.startsWith('--output='))) {
            args.push(`--output=${path.join(workDir, 'test-results')}`);
        }
        const playwright = spawn(process.execPath, [
            path.join(root, 'node_modules', '@playwright', 'test', 'cli.js'),
            'test', ...args,
        ], {
            cwd: root,
            stdio: 'inherit',
            windowsHide: true,
            env: {
                ...process.env,
                NEXTRA_E2E_PORT: String(port),
                NEXTRA_E2E_RTC_MIN_PORT: String(rtcPort),
                NEXTRA_E2E_RTC_MAX_PORT: String(rtcPort),
                // Specs that start their own server.js serve this same build.
                NEXTRA_TEST_DIST_DIR: distDir,
            },
        });
        return await new Promise((resolve) => playwright.on('exit', (code) => resolve(code ?? 1)));
    } finally {
        killTree(server);
        await sleep(300);
    }
}

main().then(
    (code) => {
        if (code === 0) cleanup();
        else console.error(`run-e2e: failed (exit ${code}); traces and videos kept in ${path.join(workDir, 'test-results')}`);
        process.exit(code);
    },
    (err) => { console.error(err.stack || err); cleanup(); process.exit(1); },
);

function cleanup() {
    // workDir is a plain temp folder made above: it holds no links.
    try { fs.rmSync(workDir, { recursive: true, force: true }); } catch { /* best effort */ }
}
